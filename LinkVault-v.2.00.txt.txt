    <script>
        // State Management
        let currentMode = 'links'; // 'links' or 'text'
        let vaultData = {
            links: [],
            texts: [],
            settings: {
                theme: 'light', // Default to light mode state
                bgImage: '',
                bgOpacity: 0.2,
                accountKey: ''
            }
        };

        // Initialize or load vault key & data
        function initVault() {
            let storedKey = localStorage.getItem('linkvault_active_key');
            if (!storedKey) {
                storedKey = generateAccountKey();
                localStorage.setItem('linkvault_active_key', storedKey);
            }
            loadVaultData(storedKey);
        }

        function generateAccountKey() {
            const chars = 'abcdefghijklmnopqrstuvwxyz0123456789';
            const segments = [4, 4, 4, 4, 2];
            return segments.map(len => {
                let seg = '';
                for (let i = 0; i < len; i++) {
                    seg += chars.charAt(Math.floor(Math.random() * chars.length));
                }
                return seg;
            }).join('-');
        }

        function getStorageKey(accountKey) {
            return `linkvault_data_${accountKey}`;
        }

        function loadVaultData(accountKey) {
            const raw = localStorage.getItem(getStorageKey(accountKey));
            if (raw) {
                try {
                    vaultData = JSON.parse(raw);
                } catch (e) {
                    console.error('Failed to parse vault data', e);
                }
            } else {
                // Default starter items for new account
                vaultData = {
                    links: [
                        { id: 1, title: 'GitHub', value: 'https://github.com', note: 'Development' },
                        { id: 2, title: 'Google', value: 'https://google.com', note: 'Search Engine' }
                    ],
                    texts: [
                        { id: 1, title: 'Python Snippet', value: 'print("Hello from LinkVault!")', note: 'ex. Python, Roast, Important text' }
                    ],
                    settings: {
                        theme: 'light',
                        bgImage: '',
                        bgOpacity: 0.2,
                        accountKey: accountKey
                    }
                };
                saveVaultData();
            }

            if (!vaultData.settings.theme) {
                vaultData.settings.theme = 'light';
            }

            vaultData.settings.accountKey = accountKey;
            document.getElementById('accountKeyInput').value = accountKey;
            applySettings();
            renderItems();
        }

        function saveVaultData() {
            const key = vaultData.settings.accountKey;
            localStorage.setItem(getStorageKey(key), JSON.stringify(vaultData));
        }

        // Mode Switching ('links' vs 'text')
        function setMode(mode) {
            currentMode = mode;
            const linksBtn = document.getElementById('modeLinksBtn');
            const textBtn = document.getElementById('modeTextBtn');
            const formTitle = document.getElementById('formTitle');
            const primaryContainer = document.getElementById('primaryInputContainer');

            if (mode === 'links') {
                linksBtn.className = "px-4 py-1.5 rounded-lg text-xs font-medium transition-all bg-indigo-600 text-white shadow";
                textBtn.className = "px-4 py-1.5 rounded-lg text-xs font-medium transition-all opacity-70 hover:opacity-100";
                formTitle.innerHTML = '<i class="fa-solid fa-plus text-indigo-400"></i> Add New Link';
                primaryContainer.innerHTML = `<input type="url" id="itemValue" placeholder="https://example.com" required class="w-full bg-current/5 border border-current/10 rounded-xl px-4 py-2.5 text-sm placeholder-current/40 focus:outline-none focus:border-indigo-500 transition-all">`;
            } else {
                textBtn.className = "px-4 py-1.5 rounded-lg text-xs font-medium transition-all bg-indigo-600 text-white shadow";
                linksBtn.className = "px-4 py-1.5 rounded-lg text-xs font-medium transition-all opacity-70 hover:opacity-100";
                formTitle.innerHTML = '<i class="fa-solid fa-code text-indigo-400"></i> Add Text or Script';
                primaryContainer.innerHTML = `<textarea id="itemValue" placeholder="Paste your text or script here..." rows="2" required class="w-full bg-current/5 border border-current/10 rounded-xl px-4 py-2.5 text-sm placeholder-current/40 focus:outline-none focus:border-indigo-500 transition-all"></textarea>`;
            }
            renderItems();
        }

        // Form Submission
        function handleFormSubmit(e) {
            e.preventDefault();
            const title = document.getElementById('itemTitle').value.trim();
            const value = document.getElementById('itemValue').value.trim();
            const note = document.getElementById('itemNote').value.trim();

            const newItem = {
                id: Date.now(),
                title,
                value,
                note
            };

            if (currentMode === 'links') {
                vaultData.links.unshift(newItem);
            } else {
                vaultData.texts.unshift(newItem);
            }

            saveVaultData();
            document.getElementById('vaultForm').reset();
            renderItems();
        }

        // Render Vault Items
        function renderItems() {
            const grid = document.getElementById('vaultGrid');
            const emptyState = document.getElementById('emptyState');
            const searchVal = document.getElementById('searchInput').value.toLowerCase();

            const items = currentMode === 'links' ? vaultData.links : vaultData.texts;
            const filtered = items.filter(item => 
                item.title.toLowerCase().includes(searchVal) || 
                item.value.toLowerCase().includes(searchVal) || 
                (item.note && item.note.toLowerCase().includes(searchVal))
            );

            grid.innerHTML = '';

            if (filtered.length === 0) {
                emptyState.classList.remove('hidden');
                emptyState.classList.add('flex');
                return;
            }

            emptyState.classList.remove('flex');
            emptyState.classList.add('hidden');

            filtered.forEach(item => {
                const card = document.createElement('div');
                card.className = "bg-current/5 border border-current/10 rounded-2xl p-4 backdrop-blur-md flex flex-col justify-between gap-3 hover:border-indigo-500/50 transition-all group";
                
                let contentHTML = '';
                if (currentMode === 'links') {
                    contentHTML = `
                        <a href="${item.value}" target="_blank" class="text-sm text-indigo-400 hover:underline truncate block flex items-center gap-2">
                            <i class="fa-solid fa-arrow-up-right-from-square text-xs opacity-70"></i> ${escapeHTML(item.value)}
                        </a>
                    `;
                } else {
                    contentHTML = `
                        <div class="bg-black/20 border border-current/5 rounded-xl p-3 font-mono text-xs max-h-28 overflow-y-auto whitespace-pre-wrap select-all">
                            ${escapeHTML(item.value)}
                        </div>
                    `;
                }

                card.innerHTML = `
                    <div>
                        <div class="flex items-center justify-between gap-2 mb-1">
                            <h3 class="font-semibold text-sm truncate">${escapeHTML(item.title)}</h3>
                            <button onclick="deleteItem(${item.id})" class="opacity-0 group-hover:opacity-100 hover:text-rose-400 transition-all p-1">
                                <i class="fa-solid fa-trash-can text-xs"></i>
                            </button>
                        </div>
                        ${contentHTML}
                    </div>
                    <div class="flex items-center justify-between pt-2 border-t border-current/5 text-xs">
                        <span class="opacity-70 truncate max-w-[200px]">${item.note ? escapeHTML(item.note) : ''}</span>
                        ${currentMode === 'text' ? `
                            <button onclick="copyText(this, \`${escapeJsString(item.value)}\`)" class="bg-current/10 hover:bg-current/20 px-3 py-1 rounded-lg text-xs font-medium transition-all flex items-center gap-1.5 shrink-0">
                                <i class="fa-solid fa-copy"></i> Copy
                            </button>
                        ` : `
                            <a href="${item.value}" target="_blank" class="bg-indigo-600/20 hover:bg-indigo-600/40 text-indigo-400 px-3 py-1 rounded-lg text-xs font-medium transition-all">
                                Open
                            </a>
                        `}
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function deleteItem(id) {
            if (currentMode === 'links') {
                vaultData.links = vaultData.links.filter(i => i.id !== id);
            } else {
                vaultData.texts = vaultData.texts.filter(i => i.id !== id);
            }
            saveVaultData();
            renderItems();
        }

        function copyText(btn, text) {
            navigator.clipboard.writeText(text).then(() => {
                const originalHTML = btn.innerHTML;
                btn.innerHTML = `<i class="fa-solid fa-check text-emerald-400"></i> Copied`;
                btn.classList.add('bg-emerald-500/20', 'text-emerald-300');
                setTimeout(() => {
                    btn.innerHTML = originalHTML;
                    btn.classList.remove('bg-emerald-500/20', 'text-emerald-300');
                }, 1500);
            });
        }

        function toggleTheme() {
            vaultData.settings.theme = vaultData.settings.theme === 'light' ? 'dark' : 'light';
            saveVaultData();
            applySettings();
        }

        function toggleSettings() {
            const modal = document.getElementById('settingsModal');
            modal.classList.toggle('hidden');
        }

        function copyAccountKey() {
            const input = document.getElementById('accountKeyInput');
            navigator.clipboard.writeText(input.value);
            event.target.innerHTML = `<i class="fa-solid fa-check"></i> Copied`;
            setTimeout(() => {
                event.target.innerHTML = `<i class="fa-solid fa-copy"></i> Copy`;
            }, 1500);
        }

        function restoreAccount() {
            const keyInput = document.getElementById('restoreKeyInput').value.trim();
            if (!keyInput) return alert('Please enter a valid account key.');
            
            localStorage.setItem('linkvault_active_key', keyInput);
            loadVaultData(keyInput);
            toggleSettings();
            document.getElementById('restoreKeyInput').value = '';
            alert('Account successfully restored!');
        }

        function handleImageUpload(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(event) {
                vaultData.settings.bgImage = event.target.result;
                saveVaultData();
                applySettings();
            };
            reader.readAsDataURL(file);
        }

        function clearBackground() {
            vaultData.settings.bgImage = '';
            saveVaultData();
            applySettings();
        }

        function updateOpacity(val) {
            const opacity = val / 100;
            vaultData.settings.bgOpacity = opacity;
            document.getElementById('opacityVal').innerText = val + '%';
            applyOverlayColor();
            saveVaultData();
        }

        function applyOverlayColor() {
            const overlay = document.getElementById('bgOverlay');
            const opacity = vaultData.settings.bgOpacity !== undefined ? vaultData.settings.bgOpacity : 0.2;
            const isDark = vaultData.settings.theme === 'dark';
            
            if (isDark) {
                overlay.style.backgroundColor = `rgba(10, 15, 30, ${opacity})`;
            } else {
                overlay.style.backgroundColor = `rgba(245, 247, 250, ${opacity})`;
            }
        }

        function applySettings() {
            const pageBody = document.getElementById('pageBody');
            const bgLayer = document.getElementById('bgLayer');
            const slider = document.getElementById('opacitySlider');
            const opacityVal = document.getElementById('opacityVal');
            const themeBtn = document.getElementById('themeToggleBtn');

            const theme = vaultData.settings.theme || 'light';

            if (theme === 'dark') {
                pageBody.className = "h-screen overflow-y-auto font-sans selection:bg-indigo-500 selection:text-white relative flex flex-col transition-colors duration-300 bg-slate-950 text-slate-100 dark";
                themeBtn.innerHTML = `<i class="fa-solid fa-moon"></i> Dark Mode`;
            } else {
                pageBody.className = "h-screen overflow-y-auto font-sans selection:bg-indigo-500 selection:text-white relative flex flex-col transition-colors duration-300 bg-slate-100 text-slate-900";
                themeBtn.innerHTML = `<i class="fa-solid fa-sun"></i> Light Mode`;
            }

            if (vaultData.settings.bgImage) {
                bgLayer.style.backgroundImage = `url('${vaultData.settings.bgImage}')`;
            } else {
                bgLayer.style.backgroundImage = 'none';
            }

            const opacity = vaultData.settings.bgOpacity !== undefined ? vaultData.settings.bgOpacity : 0.2;
            slider.value = Math.round(opacity * 100);
            opacityVal.innerText = Math.round(opacity * 100) + '%';
            applyOverlayColor();
        }

        function escapeHTML(str) {
            return str.replace(/[&<>'"]/g, 
                tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)
            );
        }

        function escapeJsString(str) {
            return str.replace(/['"\\]/g, '\\$&');
        }

        window.onload = initVault;
    </script>
